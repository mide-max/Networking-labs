enable
configure terminal
hostname R1

 no shutdown
exit

interface g0/0.10
 encapsulation dot1Q 10
 ip address 10.10.0.1 255.255.255.0
exit

interface g0/0.20
 encapsulation dot1Q 20
 ip address 10.20.0.1 255.255.255.0
exit

interface g0/0.30
 encapsulation dot1Q 30
 ip address 10.30.0.1 255.255.255.0
exit

interface g0/0.40
 encapsulation dot1Q 40
 ip address 10.40.0.1 255.255.255.0
 ip access-group GUEST-ISOLATION in
exit

interface g0/0.99
 encapsulation dot1Q 99 native
 ip address 10.99.0.1 255.255.255.0
exit

! DHCP exclusions
ip dhcp excluded-address 10.10.0.1 10.10.0.10
ip dhcp excluded-address 10.20.0.1 10.20.0.10
ip dhcp excluded-address 10.30.0.1 10.30.0.10
ip dhcp excluded-address 10.40.0.1 10.40.0.10

! DHCP pools
ip dhcp pool VLAN10-Data
 network 10.10.0.0 255.255.255.0
 default-router 10.10.0.1
 dns-server 8.8.8.8
exit

ip dhcp pool VLAN20-Voice
 network 10.20.0.0 255.255.255.0
 default-router 10.20.0.1
 dns-server 8.8.8.8
exit

ip dhcp pool VLAN30-IoT
 network 10.30.0.0 255.255.255.0
 default-router 10.30.0.1
 dns-server 8.8.8.8
exit

ip dhcp pool VLAN40-Guest
 network 10.40.0.0 255.255.255.0
 default-router 10.40.0.1
 dns-server 8.8.8.8
